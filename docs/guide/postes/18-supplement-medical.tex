\poste{18}{Supplément remboursable pour frais médicaux (fédéral)}{CA\_medic}

\pdesc Pendant fédéral du poste~12 : un crédit remboursable (ligne 45200)
pour les travailleurs à faible revenu ayant des frais médicaux élevés.
\textbf{Comme au poste~12, ce poste vaut toujours zéro dans le modèle} ---
la seule dépense médicale disponible (la prime RAMQ) ne franchit jamais le
plancher des frais admissibles.

\pobj Rembourser une partie des frais médicaux des travailleurs modestes,
même lorsqu'ils paient peu ou pas d'impôt.

\pregle Structure officielle, telle que codée :
\begin{enumerate}
  \item admissibilité : revenu de travail d'au moins $r_{min}$ ;
  \item plancher des frais : $\min(0{,}03 \times \text{revenu net},\; p)$
        (ligne 33200) ; frais admissibles
        $F_a = \pos{F - \text{plancher}}$ ;
  \item supplément avant réduction :
        $c = \min(0{,}25 \times F_a,\; c_{max})$ ;
  \item réduction : $\pos{c - 0{,}05 \times \pos{\AFNI - s}}$ --- la forme
        de réduction est ici \emph{correcte} (contrairement à l'anomalie
        du poste~12 québécois), et la base de réduction est l'$\AFNI$,
        comme pour l'ACE, le crédit TPS et l'ACT.
\end{enumerate}
